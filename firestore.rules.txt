rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function userPath(uid) {
      return /databases/$(database)/documents/users/$(uid);
    }

    function academyPath(academyId) {
      return /databases/$(database)/documents/academies/$(academyId);
    }

    function isAdmin() {
      return signedIn()
        && exists(userPath(request.auth.uid))
        && get(userPath(request.auth.uid)).data.role == "admin"
        && get(userPath(request.auth.uid)).data.isActive == true;
    }

    function isAcademyAdmin(academyId) {
      return isAdmin()
        && get(userPath(request.auth.uid)).data.academyId == academyId;
    }

    function isActiveStudent() {
      return signedIn()
        && exists(userPath(request.auth.uid))
        && get(userPath(request.auth.uid)).data.role == "student"
        && get(userPath(request.auth.uid)).data.isActive == true;
    }

    function isAcademyStudent(academyId) {
      return isActiveStudent()
        && get(userPath(request.auth.uid)).data.academyId == academyId;
    }

    function isActiveTeacher() {
      return signedIn()
        && exists(userPath(request.auth.uid))
        && get(userPath(request.auth.uid)).data.role == "teacher"
        && get(userPath(request.auth.uid)).data.isActive == true;
    }

    function isAcademyTeacher(academyId) {
      return isActiveTeacher()
        && get(userPath(request.auth.uid)).data.academyId == academyId;
    }

    function userProfile() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function activeAdminFor(academyId) {
      return signedIn()
        && userProfile().role == "admin"
        && userProfile().isActive == true
        && userProfile().academyId == academyId;
    }

    function activeStudentFor(academyId) {
      return signedIn()
        && userProfile().role == "student"
        && userProfile().isActive == true
        && userProfile().academyId == academyId;
    }

    match /studyMaterials/{materialId} {
      // Admin, teacher, and active student in the same academy can read study materials
      allow read: if isAcademyAdmin(resource.data.academyId)
        || isAcademyTeacher(resource.data.academyId)
        || isAcademyStudent(resource.data.academyId);

      // Teacher in the academy can create study materials for their academy
      allow create: if isAcademyTeacher(request.resource.data.academyId)
        && request.resource.data.uploadedBy == request.auth.uid
        && request.resource.data.classId is string
        && request.resource.data.subject is string
        && request.resource.data.title is string
        && request.resource.data.sourceType in ["drive-file", "external-video"]
        && request.resource.data.createdAt == request.time;

      // Teacher who uploaded or academy admin can update/delete
      allow update: if (
          isAcademyTeacher(resource.data.academyId)
          && resource.data.uploadedBy == request.auth.uid
        ) || isAcademyAdmin(resource.data.academyId);

      allow delete: if (
          isAcademyTeacher(resource.data.academyId)
          && resource.data.uploadedBy == request.auth.uid
        ) || isAcademyAdmin(resource.data.academyId);
    }

    match /academies/{academyId} {
      allow read: if resource.data.status == "active";

      allow create: if signedIn()
        && academyId == request.auth.uid
        && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.status == "active"
        && getAfter(userPath(request.auth.uid)).data.role == "admin"
        && getAfter(userPath(request.auth.uid)).data.isActive == true
        && getAfter(userPath(request.auth.uid)).data.academyId == academyId;

      allow update: if isAcademyAdmin(academyId)
        && request.resource.data.ownerUid == resource.data.ownerUid
        && request.resource.data.status == resource.data.status
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "name",
            "email",
            "phone",
            "address",
            "settings",
            "updatedAt"
          ]);

      allow delete: if false;
    }

    match /syllabus/{syllabusId} {
      function syllabusClassExists(classId) {
        return exists(
          /databases/$(database)/documents/classes/$(classId)
        );
      }

      function syllabusClassData(classId) {
        return get(
          /databases/$(database)/documents/classes/$(classId)
        ).data;
      }

      allow read: if
        isAcademyAdmin(resource.data.academyId)
        ||
        (
          isAcademyTeacher(resource.data.academyId)
          && resource.data.teacherId == request.auth.uid
        );

      allow create: if
        (isAcademyAdmin(request.resource.data.academyId) || isAcademyTeacher(request.resource.data.academyId))
        && request.resource.data.classId is string
        && syllabusClassExists(request.resource.data.classId)
        && syllabusClassData(request.resource.data.classId).academyId == request.resource.data.academyId
        && request.resource.data.className is string
        && request.resource.data.subject is string
        && request.resource.data.chapterTitle is string
        && request.resource.data.topics is list;

      allow update: if
        isAcademyAdmin(resource.data.academyId)

        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.classId == resource.data.classId
        && request.resource.data.teacherId == resource.data.teacherId
        && request.resource.data.subject == resource.data.subject

        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "chapterTitle",
            "topics",
            "updatedAt"
          ])

        && request.resource.data.chapterTitle is string
        && request.resource.data.topics is list;

      allow update: if
        isAcademyTeacher(resource.data.academyId)

        && resource.data.teacherId == request.auth.uid

        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.classId == resource.data.classId
        && request.resource.data.teacherId == resource.data.teacherId
        && request.resource.data.subject == resource.data.subject

        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "chapterTitle",
            "topics",
            "className",
            "teacherName",
            "isCompleted",
            "updatedAt"
          ])

        && request.resource.data.chapterTitle is string
        && request.resource.data.topics is list

        && (
          !("isCompleted" in request.resource.data)
          || request.resource.data.isCompleted is bool
        );

      allow delete: if
        isAcademyAdmin(resource.data.academyId);

      allow delete: if
        isAcademyTeacher(resource.data.academyId)
        && resource.data.teacherId == request.auth.uid;
    }

    match /users/{uid} {
      allow get: if signedIn()
        && (
          request.auth.uid == uid
          || isAdmin()
        );

      allow list: if isAdmin();

      allow get, list: if signedIn()
        && isAcademyTeacher(resource.data.academyId)
        && resource.data.role == "student"
        && resource.data.isActive == true;

      allow create: if signedIn()
        && request.auth.uid == uid
        && request.resource.data.email == request.auth.token.email
        && (
          (
            request.resource.data.role == "admin"
            && request.resource.data.isActive == true
            && request.resource.data.academyId == uid
            && request.resource.data.provisionalAdmin == true
            && getAfter(academyPath(uid)).data.ownerUid == uid
          )
          ||
          (
            request.resource.data.role == "pending"
            && request.resource.data.isActive == false
            && request.resource.data.registrationStatus == "pending"
            && request.resource.data.requestedRole in ["student", "teacher"]
            && request.resource.data.academyId is string
            && exists(academyPath(request.resource.data.academyId))
            && get(academyPath(request.resource.data.academyId)).data.status == "active"
          )
        );

      allow update: if isAcademyAdmin(resource.data.academyId)
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "role",
            "isActive",
            "registrationStatus",
            "reviewedBy",
            "reviewedAt"
          ])
        && request.resource.data.role in ["student", "teacher"]
        && request.resource.data.isActive == true
        && request.resource.data.reviewedBy == request.auth.uid;

      allow update: if signedIn()
        && request.auth.uid == uid
        && resource.data.role == "admin"
        && resource.data.isActive == true
        && resource.data.provisionalAdmin == true
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "provisionalAdmin",
            "activityCompleted",
            "onboardingCompletedAt",
            "studentCountAtCompletion",
            "teacherCountAtCompletion"
          ])
        && request.resource.data.provisionalAdmin == false
        && request.resource.data.activityCompleted == true
        && request.resource.data.studentCountAtCompletion is int
        && request.resource.data.studentCountAtCompletion >= 2
        && request.resource.data.teacherCountAtCompletion is int
        && request.resource.data.teacherCountAtCompletion >= 1
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.role == resource.data.role
        && request.resource.data.isActive == resource.data.isActive;

      allow update: if signedIn()
        && request.auth.uid == uid
        && resource.data.role == "admin"
        && resource.data.provisionalAdmin == true
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "isActive",
            "registrationStatus",
            "restrictionReason",
            "restrictedAt"
          ])
        && request.resource.data.isActive == false
        && request.resource.data.registrationStatus == "expired"
        && request.resource.data.role == resource.data.role
        && request.resource.data.academyId == resource.data.academyId;

      allow update: if isAcademyAdmin(resource.data.academyId)
        && resource.data.role == "student"
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "fullName",
            "email",
            "phone",
            "classId",
            "className",
            "classIds",
            "subjects",
            "enrollmentStatus",
            "isActive",
            "statusUpdatedAt",
            "statusUpdatedBy"
          ])
        && request.resource.data.fullName is string
        && request.resource.data.email is string
        && request.resource.data.enrollmentStatus
          in ["active", "left", "terminated"]
        && request.resource.data.isActive
          == (request.resource.data.enrollmentStatus == "active")
        && request.resource.data.statusUpdatedBy == request.auth.uid
        && request.resource.data.statusUpdatedAt == request.time;

      allow update: if signedIn()
        && request.auth.uid == uid
        && resource.data.role == "admin"
        && resource.data.isActive == true
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly(["fullName", "phone", "updatedAt"])
        && request.resource.data.fullName is string
        && request.resource.data.phone is string
        && request.resource.data.role == resource.data.role
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.isActive == resource.data.isActive;

      allow update: if isAcademyAdmin(resource.data.academyId)
        && resource.data.role == "teacher"
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "isActive",
            "statusUpdatedAt",
            "statusUpdatedBy"
          ])
        && request.resource.data.statusUpdatedBy == request.auth.uid
        && request.resource.data.statusUpdatedAt == request.time;

      allow delete: if false;
    }

    match /registrationRequests/{uid} {
      allow create: if signedIn()
        && request.auth.uid == uid
        && request.resource.data.uid == uid
        && request.resource.data.requestedRole in ["student", "teacher"]
        && request.resource.data.status == "pending"
        && request.resource.data.academyId is string
        && exists(academyPath(request.resource.data.academyId))
        && get(academyPath(request.resource.data.academyId)).data.status == "active"
        && getAfter(userPath(uid)).data.role == "pending"
        && getAfter(userPath(uid)).data.academyId == request.resource.data.academyId;

      allow get: if signedIn()
        && (
          request.auth.uid == uid
          || isAcademyAdmin(resource.data.academyId)
        );

      allow list: if isAdmin()
        && resource.data.academyId
          == get(userPath(request.auth.uid)).data.academyId;

      allow update: if isAcademyAdmin(resource.data.academyId)
        && resource.data.status == "pending"
        && request.resource.data.status in ["approved", "rejected"]
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly(["status", "reviewedBy", "reviewedAt"])
        && request.resource.data.reviewedBy == request.auth.uid;

      allow delete: if false;
    }

    match /activityLogs/{logId} {
      allow create: if isAcademyAdmin(request.resource.data.academyId)
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.action in [
          "student_created",
          "teacher_created",
          "class_created",
          "class_updated",
          "class_deleted",
          "attendance_saved",
          "student_status_updated",
          "fee_created",
          "fee_updated",
          "payment_reviewed"
        ]
        && request.resource.data.title is string
        && request.resource.data.details is string
        && request.resource.data.actionType is string
        && request.resource.data.createdAt == request.time;

      allow read: if isAcademyAdmin(resource.data.academyId);

      allow update, delete: if false;
    }

    match /classes/{classId} {
      allow read: if resource.data.status == "active"
        || isAcademyAdmin(resource.data.academyId)
        || isAcademyStudent(resource.data.academyId)
        || isAcademyTeacher(resource.data.academyId);

      allow create: if isAdmin()
        && request.resource.data.academyId
          == get(userPath(request.auth.uid)).data.academyId
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.className is string
        && request.resource.data.status in ["active", "inactive"]
        && request.resource.data.subjects is list;

      allow update: if isAcademyAdmin(resource.data.academyId)
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.createdBy == resource.data.createdBy
        && request.resource.data.className is string
        && request.resource.data.status in ["active", "inactive"]
        && request.resource.data.subjects is list;

      allow delete: if isAcademyAdmin(resource.data.academyId);
    }

    match /attendance/{attendanceId} {
      function classExists(classId) {
        return exists(
          /databases/$(database)/documents/classes/$(classId)
        );
      }

      function classData(classId) {
        return get(
          /databases/$(database)/documents/classes/$(classId)
        ).data;
      }

      // Helper: verify teacher is assigned to this class (at class level or via subjects)
      function isAssignedTeacherForClass(classId) {
        return classExists(classId)
          && (
            classData(classId).teacherId == request.auth.uid
            || (
              classData(classId).subjects is list
              && classData(classId).subjects.size() > 0
            )
          );
      }

      function validAttendanceId(data) {
        return attendanceId == data.classId + "_" + data.date + "_" + data.studentId;
      }

      // READ: Admin and teacher in the same academy; student only their own records
      allow read: if isAcademyAdmin(resource.data.academyId)
        || isAcademyTeacher(resource.data.academyId)
        || (
          isAcademyStudent(resource.data.academyId)
          && resource.data.studentId == request.auth.uid
        );

      // ALLOW GET FOR DUPLICATE CHECK BEFORE CREATE
      allow get: if (isActiveStudent() && attendanceId.matches('^.*_' + request.auth.uid + '$'))
        || isActiveTeacher()
        || isAdmin();

      // CREATE (Student): Self-mark once per class per day if not already marked
      allow create: if isAcademyStudent(request.resource.data.academyId)
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.markedBy == request.auth.uid
        && request.resource.data.markedByRole == "student"
        && request.resource.data.classId is string
        && classExists(request.resource.data.classId)
        && classData(request.resource.data.classId).academyId == request.resource.data.academyId
        && classData(request.resource.data.classId).status == "active"
        && request.resource.data.className is string
        && request.resource.data.date is string
        && request.resource.data.studentName is string
        && request.resource.data.status == "present"
        && request.resource.data.originalStatus == "present"
        && request.resource.data.punchedAt == request.time
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time
        && request.resource.data.editHistory is list
        && request.resource.data.editHistory.size() == 0
        && validAttendanceId(request.resource.data)
        && request.resource.data.keys().hasOnly([
          "academyId",
          "classId",
          "className",
          "date",
          "studentId",
          "studentName",
          "status",
          "originalStatus",
          "markedBy",
          "markedByRole",
          "punchedAt",
          "createdAt",
          "updatedAt",
          "editHistory"
        ]);

      // CREATE (Teacher): Mark attendance for assigned class only (if not already marked)
      allow create: if isAcademyTeacher(request.resource.data.academyId)
        && request.resource.data.teacherId == request.auth.uid
        && request.resource.data.markedBy == request.auth.uid
        && request.resource.data.markedByRole == "teacher"
        && request.resource.data.classId is string
        && isAssignedTeacherForClass(request.resource.data.classId)
        && classData(request.resource.data.classId).academyId == request.resource.data.academyId
        && classData(request.resource.data.classId).status == "active"
        && request.resource.data.studentId is string
        && request.resource.data.date is string
        && request.resource.data.status in ["present", "absent", "late", "excused"]
        && request.resource.data.originalStatus == request.resource.data.status
        && request.resource.data.punchedAt == request.time
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time
        && request.resource.data.editHistory is list
        && validAttendanceId(request.resource.data);

      // UPDATE (Teacher only): Edit status and append to editHistory for assigned class
      allow update: if isAcademyTeacher(resource.data.academyId)
        && isAssignedTeacherForClass(resource.data.classId)
        && classData(resource.data.classId).academyId == resource.data.academyId
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.classId == resource.data.classId
        && request.resource.data.studentId == resource.data.studentId
        && request.resource.data.date == resource.data.date
        && request.resource.data.punchedAt == resource.data.punchedAt
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.originalStatus == resource.data.originalStatus
        && request.resource.data.markedBy == resource.data.markedBy
        && request.resource.data.markedByRole == resource.data.markedByRole
        && request.resource.data.teacherId == request.auth.uid
        && request.resource.data.status in ["present", "absent", "late", "excused"]
        && request.resource.data.editHistory is list
        && request.resource.data.editHistory.size() >= resource.data.editHistory.size()
        && request.resource.data.updatedAt == request.time
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "status",
          "originalStatus",
          "teacherId",
          "teacherName",
          "updatedAt",
          "editHistory"
        ]);

      // DELETE: Admin only
      allow delete: if isAcademyAdmin(resource.data.academyId);
    }

    match /schedules/{scheduleId} {
      allow read: if isAcademyAdmin(resource.data.academyId)
        || (
          isAcademyTeacher(resource.data.academyId)
          && resource.data.teacherId == request.auth.uid
        );

      allow create: if isAcademyAdmin(request.resource.data.academyId)
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.classId is string
        && request.resource.data.teacherId is string
        && request.resource.data.title is string
        && request.resource.data.subject is string
        && request.resource.data.date is string
        && request.resource.data.startTime is string
        && request.resource.data.endTime is string
        && request.resource.data.status in ["scheduled", "cancelled"]
        && exists(/databases/$(database)/documents/classes/$(request.resource.data.classId))
        && get(/databases/$(database)/documents/classes/$(request.resource.data.classId)).data.academyId
          == request.resource.data.academyId
        && get(/databases/$(database)/documents/classes/$(request.resource.data.classId)).data.status
          == "active"
        && exists(userPath(request.resource.data.teacherId))
        && get(userPath(request.resource.data.teacherId)).data.academyId
          == request.resource.data.academyId
        && get(userPath(request.resource.data.teacherId)).data.role == "teacher"
        && get(userPath(request.resource.data.teacherId)).data.isActive == true;

      allow update: if isAcademyAdmin(resource.data.academyId)
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.createdBy == resource.data.createdBy
        && request.resource.data.classId is string
        && request.resource.data.teacherId is string
        && request.resource.data.title is string
        && request.resource.data.subject is string
        && request.resource.data.date is string
        && request.resource.data.startTime is string
        && request.resource.data.endTime is string
        && request.resource.data.status in ["scheduled", "cancelled"]
        && exists(/databases/$(database)/documents/classes/$(request.resource.data.classId))
        && get(/databases/$(database)/documents/classes/$(request.resource.data.classId)).data.academyId
          == resource.data.academyId
        && exists(userPath(request.resource.data.teacherId))
        && get(userPath(request.resource.data.teacherId)).data.academyId
          == resource.data.academyId
        && get(userPath(request.resource.data.teacherId)).data.role == "teacher"
        && get(userPath(request.resource.data.teacherId)).data.isActive == true;

      allow delete: if isAcademyAdmin(resource.data.academyId);
    }

    match /liveClasses/{liveClassId} {
      // Admin, teacher, and active students in the same academy can read live classes
      allow read: if isAcademyAdmin(resource.data.academyId)
        || isAcademyTeacher(resource.data.academyId)
        || isAcademyStudent(resource.data.academyId);

      // Assigned teacher can start or update their own live class session
      allow create, update: if isAcademyTeacher(request.resource.data.academyId)
        && request.resource.data.teacherId == request.auth.uid
        && request.resource.data.classId is string
        && request.resource.data.className is string
        && request.resource.data.subject is string
        && request.resource.data.mode in ["jitsi", "youtube"]
        && request.resource.data.status in ["live", "ended"]
        && request.resource.data.updatedAt == request.time;

      allow delete: if isAcademyAdmin(resource.data.academyId)
        || (
          isAcademyTeacher(resource.data.academyId)
          && resource.data.teacherId == request.auth.uid
        );
    }

    match /fees/{feeId} {
      allow get, list: if isAcademyAdmin(resource.data.academyId)
        || (
          signedIn()
          && resource.data.studentId == request.auth.uid
          && get(userPath(request.auth.uid)).data.role == "student"
          && get(userPath(request.auth.uid)).data.academyId
            == resource.data.academyId
        );

      allow create: if isAcademyAdmin(request.resource.data.academyId)
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.studentId is string
        && request.resource.data.studentName is string
        && request.resource.data.feeType is string
        && request.resource.data.amount is number
        && request.resource.data.amount > 0
        && request.resource.data.amountPaid == 0
        && request.resource.data.dueDate is string
        && request.resource.data.status == "pending"
        && exists(userPath(request.resource.data.studentId))
        && get(userPath(request.resource.data.studentId)).data.role
          == "student"
        && get(userPath(request.resource.data.studentId)).data.isActive
          == true
        && get(userPath(request.resource.data.studentId)).data.academyId
          == request.resource.data.academyId;

      allow update: if isAcademyAdmin(resource.data.academyId)
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.studentId == resource.data.studentId
        && request.resource.data.amount == resource.data.amount
        && request.resource.data.createdBy == resource.data.createdBy
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(["amountPaid", "status", "updatedAt", "installments"])
        && request.resource.data.amountPaid is number
        && request.resource.data.amountPaid >= resource.data.amountPaid
        && request.resource.data.amountPaid <= resource.data.amount
        && request.resource.data.status in ["pending", "partial", "paid", "unpaid", "pending_review"]
        && (
          !("installments" in request.resource.data)
          || request.resource.data.installments is list
        );

      allow delete: if false;
    }

    match /paymentSubmissions/{paymentId} {
      allow read: if signedIn()
        && (
          isAcademyAdmin(resource.data.academyId)
          || (
            activeStudentFor(resource.data.academyId)
            && resource.data.studentId == request.auth.uid
          )
        );

      allow create: if signedIn()
        && activeStudentFor(request.resource.data.academyId)
        && request.resource.data.studentId == request.auth.uid
        && request.resource.data.status in ["pending", "pending_review"]
        && request.resource.data.amount is number
        && request.resource.data.amount > 0
        && request.resource.data.feeId is string
        && exists(
          /databases/$(database)/documents/fees/$(request.resource.data.feeId)
        )
        && get(
          /databases/$(database)/documents/fees/$(request.resource.data.feeId)
        ).data.academyId == request.resource.data.academyId
        && get(
          /databases/$(database)/documents/fees/$(request.resource.data.feeId)
        ).data.studentId == request.auth.uid
        && (
          (
            !("installmentNumber" in request.resource.data)
            && get(
              /databases/$(database)/documents/fees/$(request.resource.data.feeId)
            ).data.paymentPlan != "installments"
          )
          ||
          (
            request.resource.data.installmentNumber is int
            && request.resource.data.installmentNumber >= 1
            && get(
              /databases/$(database)/documents/fees/$(request.resource.data.feeId)
            ).data.paymentPlan == "installments"
            && request.resource.data.installmentNumber <= get(
              /databases/$(database)/documents/fees/$(request.resource.data.feeId)
            ).data.installmentCount
          )
        );

      allow update: if signedIn()
        && isAcademyAdmin(resource.data.academyId)
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.studentId == resource.data.studentId
        && request.resource.data.feeId == resource.data.feeId
        && request.resource.data.status in ["approved", "rejected"]
        && request.resource.data.reviewedBy == request.auth.uid;

      allow delete: if signedIn()
        && isAcademyAdmin(resource.data.academyId);
    }

    match /notifications/{notificationId} {
      function announcementClassExists(classId) {
        return exists(
          /databases/$(database)/documents/classes/$(classId)
        );
      }

      function announcementClassData(classId) {
        return get(
          /databases/$(database)/documents/classes/$(classId)
        ).data;
      }

      function studentBelongsToAnnouncementClass(classId) {
  return isAcademyStudent(resource.data.academyId)
    && announcementClassExists(classId)
    && announcementClassData(classId).academyId
      == resource.data.academyId
    && announcementClassData(classId).status == "active"
    && (
      (
        announcementClassData(classId).studentIds is list
        && announcementClassData(classId).studentIds.hasAny([request.auth.uid])
      )
      || userProfile().classId == classId
      || (
        userProfile().classIds is list
        && userProfile().classIds.hasAny([classId])
      )
      || (
        userProfile().className is string
        && userProfile().className == announcementClassData(classId).className
      )
    );
}

      allow get, list: if
        resource.data.recipientRole == "student"
        && resource.data.type == "announcement"
        && resource.data.targetType == "class"
        && resource.data.classId is string
        && studentBelongsToAnnouncementClass(
          resource.data.classId
        );

      allow create: if isAcademyTeacher(request.resource.data.academyId)
        && request.resource.data.teacherId == request.auth.uid
        && request.resource.data.recipientRole == "student"
        && request.resource.data.type == "announcement"
        && request.resource.data.title is string
        && request.resource.data.message is string
        && request.resource.data.announcementDate is string
        && request.resource.data.teacherName is string
        && request.resource.data.read == false
        && request.resource.data.isRead == false
        && request.resource.data.createdAt == request.time
        && (
          (
            request.resource.data.targetType == "academy"
            && request.resource.data.classId == null
            && request.resource.data.className == null
            && request.resource.data.subject == null
          )
          ||
          (
            request.resource.data.targetType == "class"
            && request.resource.data.classId is string
            && announcementClassExists(request.resource.data.classId)
            && announcementClassData(request.resource.data.classId).academyId
              == request.resource.data.academyId
            && announcementClassData(request.resource.data.classId).status
              == "active"
            && announcementClassData(request.resource.data.classId).className
              == request.resource.data.className
            && request.resource.data.subject is string
          )
        );

      allow get, list: if isAcademyTeacher(resource.data.academyId)
        && resource.data.teacherId == request.auth.uid
        && resource.data.recipientRole == "student"
        && resource.data.type == "announcement";

      allow update: if isAcademyTeacher(resource.data.academyId)
        && resource.data.teacherId == request.auth.uid
        && resource.data.type == "announcement"
        && resource.data.recipientRole == "student"
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.teacherId == resource.data.teacherId
        && request.resource.data.type == resource.data.type
        && request.resource.data.recipientRole == resource.data.recipientRole
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            "title",
            "message",
            "announcementDate",
            "targetType",
            "classId",
            "className",
            "subject"
          ])
        && request.resource.data.title is string
        && request.resource.data.message is string
        && request.resource.data.announcementDate is string
        && (
          (
            request.resource.data.targetType == "academy"
            && request.resource.data.classId == null
            && request.resource.data.className == null
            && request.resource.data.subject == null
          )
          ||
          (
            request.resource.data.targetType == "class"
            && request.resource.data.classId is string
            && announcementClassExists(request.resource.data.classId)
            && announcementClassData(request.resource.data.classId).academyId
              == resource.data.academyId
            && announcementClassData(request.resource.data.classId).status
              == "active"
            && request.resource.data.className is string
            && request.resource.data.subject is string
          )
        );

      allow delete: if isAcademyTeacher(resource.data.academyId)
        && resource.data.teacherId == request.auth.uid
        && resource.data.type == "announcement"
        && resource.data.recipientRole == "student";

      allow get, list: if isAcademyAdmin(resource.data.academyId)
        && resource.data.recipientRole == "student"
        && resource.data.type == "announcement";

      allow get, list: if isAcademyStudent(resource.data.academyId)
        && resource.data.studentId == request.auth.uid;

      allow get, list: if isAcademyStudent(resource.data.academyId)
        && resource.data.recipientRole == "student"
        && resource.data.type == "announcement"
        && resource.data.targetType == "academy";

      allow update: if isAcademyStudent(resource.data.academyId)
        && resource.data.studentId == request.auth.uid
        && request.resource.data.academyId == resource.data.academyId
        && request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly(["read", "isRead", "readAt"])
        && request.resource.data.read == true
        && request.resource.data.isRead == true
        && request.resource.data.readAt == request.time;

      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
